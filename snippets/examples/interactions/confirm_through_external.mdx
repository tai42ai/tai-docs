# Wire an external-confirmation ask end to end. Three pieces compose it:
#   * the stripe webhook verifier, under the canonical webhook_verifier_modules

  - tai42_webhook_verifier_stripe

extensions_modules:
  - tai42_skeleton.extensions.builtin.ask_external

    extensions:
      request_confirmation:
        - - name: ask_external
            config:
              verifier:
                name: shared_secret
